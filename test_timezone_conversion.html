測試 1: localToUTC 函數

輸入台北時間，轉換為 UTC

轉換邏輯程式碼

const TAIPEI_OFFSET_HOURS = 8

function localToUTC(dateString, timeString) {
  // Parse the date and time components
  const [year, month, day] = dateString.split('-').map(Number)
  const [hours, minutes] = timeString.split(':').map(Number)

  // Create a date in UTC, then manually adjust for Taipei timezone
  // Since user input is Taipei time (UTC+8), we subtract 8 hours to get UTC
  const utcDate = new Date(Date.UTC(
    year,
    month - 1,
    day,
    hours - TAIPEI_OFFSET_HOURS,
    minutes,
    0,
    0
  ))

  return utcDate.toISOString()
}
